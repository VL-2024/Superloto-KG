  function ensureAudio() {
    if (!audioCtx) {
      try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch (e) {}
    }
    if (audioCtx && audioCtx.state === 'suspended') audioCtx.resume();
  }
  function tone(freq, dur, type='sine', gain=0.03, when=0) {
    if (!audioCtx) return;
    const o = audioCtx.createOscillator();
    const g = audioCtx.createGain();
    const now = audioCtx.currentTime + when;
    o.type = type;
    o.frequency.setValueAtTime(freq, now);
    g.gain.setValueAtTime(0.0001, now);
    g.gain.exponentialRampToValueAtTime(gain, now + 0.01);
    g.gain.exponentialRampToValueAtTime(0.0001, now + dur);
    o.connect(g).connect(audioCtx.destination);
    o.start(now);
    o.stop(now + dur + 0.02);
  }

  function scenarioFromLms(id) {
    const scenarioId = Number(id);
    const map = {
      1: { n: 0, khan: false },
      2: { n: 1, khan: false },

    };

    return {
      id: scenarioId,

      khan: base.khan,
      guided: false,
      launched: false,
      khanStarted: false,
      targets: [],
      forcedDone: false
    };
  }

  function fmtMultiplier(n) {
    const x = Number(n || 0);
    if (Number.isInteger(x)) return String(x);
    return String(Math.round(x * 100) / 100).replace('.', ',');
  }

  function rand(a, b) { return a + Math.random() * (b - a); }
  function easeOutCubic(t) { return 1 - Math.pow(1 - t, 3); }
  function cubicPoint(t, p0, p1, p2, p3) {
    const mt = 1 - t;
    return mt*mt*mt*p0 + 3*mt*mt*t*p1 + 3*mt*t*t*p2 + t*t*t*p3;
  }

  function applyGameSettings(settings = {}) {
    // language and currency are authoritative LMS settings.
    language = normalizeLanguage(settings.language || language);
    currency = String(settings.currency || currency || 'KGS').toUpperCase();
    currencyDisplay = String(
      settings.currencyDisplay ||
      settings.currencyLabel ||
      settings.currencySymbol ||
      currency
    );

    gameMode = String(settings.mode || gameMode || 'real').toLowerCase() === 'demo'
      ? 'demo'
      : 'real';
    demoAllowed = Boolean(settings.demoAllowed);

    if (Number.isFinite(Number(settings.demoBalance))) {
      demoBalance = Number(settings.demoBalance);
    }

    // LMS is authoritative. Demo cannot be active unless LMS permits it.
    if (!demoAllowed && gameMode === 'demo') {
      gameMode = 'real';
    }

    if (Array.isArray(settings.denominations) && settings.denominations.length) {
      denominations = settings.denominations
        .map(Number)
        .filter(n => Number.isFinite(n) && n > 0);
    }

    const allowed = currentDenominations();
    const requestedStake = Number(settings.denomination ?? stake);
    stake = allowed.includes(requestedStake)
      ? requestedStake
      : (allowed[0] || requestedStake || 100);

    selectorOpen = null;
  }

  async function refreshBalance() {
    state = 'loading';

    try {
      if (gameMode === 'demo') {
        balance = demoBalance;
        state = 'idle';

        window.X2LMS.emit('X2_GAME_BALANCE_LOADED', {
          gameId: GAME_ID,
          balance,
          currency,
          currencyDisplay,
          language,
          denominations: currentDenominations(),
          mode: gameMode
        });
        return;
      }

      const data = await window.X2LMS.getBalance({currency});
      realBalance = Number(data.balance);
      balance = realBalance;

      if (data.currency) currency = String(data.currency).toUpperCase();
      if (data.currencyDisplay || data.currencyLabel || data.currencySymbol) {
        currencyDisplay = String(
          data.currencyDisplay ||
          data.currencyLabel ||
          data.currencySymbol
        );
      }

      state = 'idle';

      window.X2LMS.emit('X2_GAME_BALANCE_LOADED', {
        gameId: GAME_ID,
        balance,
        currency,
        currencyDisplay,
        language,
        denominations: currentDenominations(),
        mode: gameMode
      });
    } catch (err) {
      console.error('[CHUKO] balance load failed', err);
      state = 'error';
      showToast(tr('balanceError'), false);

      if (window.X2LMS) {
        window.X2LMS.emit('X2_GAME_ERROR', {
          stage: 'balance',
          code: err.code || 'BALANCE_ERROR',
          message: err.message || String(err)
        });
      }
    }
  }

  async function initializeLms() {
    state = 'loading';
    try {
      if (!window.X2LMS) throw new Error('LMS_ADAPTER_NOT_LOADED');

      const settings = await window.X2LMS.getGameSettings();
      applyGameSettings(settings);
      await refreshBalance();
    } catch (err) {
      console.error('[CHUKO] LMS init failed', err);
      state = 'error';
      showToast(tr('balanceError'), false);
      if (window.X2LMS) {
        window.X2LMS.emit('X2_GAME_ERROR', {
          stage: 'init',
          code: err.code || 'INIT_ERROR',
          message: err.message || String(err)
        });
      }
    }
  }

  function selectDenomination(value) {
    if (!['idle','settled'].includes(state)) return;
    const n = Number(value);
    if (!currentDenominations().includes(n)) return;

    stake = n;
    selectorOpen = null;

    window.X2LMS.emit('X2_GAME_DENOMINATION_CHANGED', {
      gameId: GAME_ID,
      denomination: stake,
      currency,
      language
    });
  }

  function requestDeposit() {
    if (!window.X2LMS) return;

    window.X2LMS.emit('X2_GAME_DEPOSIT_REQUEST', {
      gameId: GAME_ID,
      mode: gameMode,
      currency,
      denomination: stake,
      language,
      balance
    });
  }

  async function setGameMode(nextMode) {
    if (!demoAllowed) return;
    if (!['idle', 'settled'].includes(state)) return;

    const normalized = String(nextMode || '').toLowerCase() === 'demo' ? 'demo' : 'real';
    if (normalized === gameMode) return;

    gameMode = normalized;
    selectorOpen = null;
    winBubble.show = false;

    window.X2LMS.emit('X2_GAME_MODE_CHANGED', {
      gameId: GAME_ID,
      mode: gameMode,
      currency,
      language,
      denomination: stake
    });

    await refreshBalance();
  }
