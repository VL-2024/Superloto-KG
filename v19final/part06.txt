Style = p.gold ? '#FFE17F' : '#FFF1BF';
      ctx.beginPath();
      ctx.arc(p.x * S, p.y * S, p.r * S, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
  }
  function drawRipples(dt) {
    for (let i = ripples.length - 1; i >= 0; i--) {
      const r = ripples[i];
      r.life += dt;
      if (r.life >= r.maxLife) { ripples.splice(i, 1); continue; }
      const t = r.life / r.maxLife;
      const rr0 = r.r + (r.maxR - r.r) * t;
      ctx.save();
      ctx.globalAlpha = 1 - t;
      ctx.strokeStyle = r.gold ? 'rgba(255,225,120,.75)' : 'rgba(255,255,255,.38)';
      ctx.lineWidth = (r.gold ? 3 : 2) * S * (1 - t * 0.45);
      ctx.beginPath();
      ctx.arc(r.x * S, r.y * S, rr0 * S, 0, Math.PI * 2);
      ctx.stroke();
      ctx.restore();
    }
  }
  function drawConfetti(dt) {
    for (let i = confetti.length - 1; i >= 0; i--) {
      const c = confetti[i];
      c.life -= dt;
      c.x += c.vx * dt * 0.06;
      c.y += c.vy * dt * 0.06;
      c.rot += c.vr * dt * 0.03;
      if (c.life <= 0) { confetti.splice(i, 1); continue; }
      ctx.save();
      ctx.globalAlpha = Math.max(0, c.life / 1700);
      ctx.translate(c.x * S, c.y * S);
      ctx.rotate(c.rot);
      ctx.fillStyle = c.color;
      ctx.fillRect(-(c.s * S) / 2, -(c.s * S) / 2, c.s * S, c.s * S * 0.74);
      ctx.restore();
    }
  }
